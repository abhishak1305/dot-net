# C# Console Syntax

## String Methods

```csharp
string s = "hello world";

Console.WriteLine(s.ToUpper());
Console.WriteLine(s.ToLower());
Console.WriteLine(s.Length);
Console.WriteLine(s.Trim());
Console.WriteLine(s.Contains("world"));
Console.WriteLine(s.Replace("world", "C#"));
Console.WriteLine(s.StartsWith("hello"));
Console.WriteLine(s.EndsWith("world"));
```

## Input and Output

```csharp
Console.Write("Enter an integer: ");
int a = Convert.ToInt32(Console.ReadLine());

Console.Write("Enter a double: ");
double b = Convert.ToDouble(Console.ReadLine());

Console.Write("Enter your name: ");
string name = Console.ReadLine();

Console.WriteLine("Hello, " + name);
Console.WriteLine("Addition: " + (a + b));
Console.WriteLine("Subtraction: " + (a - b));
Console.WriteLine("Multiplication: " + (a * b));
Console.WriteLine("Division: " + (a / b));
```

## Switch

```csharp
string x = Console.ReadLine();

switch (x)
{
    case "A":
        Console.WriteLine("A");
        break;

    default:
        Console.WriteLine("Not A");
        break;
}
```

## For Loop

```csharp
for (int i = 0; i < 5; i++)
{
    Console.Write(i + " ");
}

Console.WriteLine();
```

## While Loop

```csharp
int i = 5;

while (i > 0)
{
    Console.Write(i + " ");
    i--;
}

Console.WriteLine();
```

## Continue

```csharp
for (int i = 0; i < 5; i++)
{
    if (i % 2 != 0)
        continue;

    Console.Write(i + " ");
}

Console.WriteLine();
```

## Break

```csharp
for (int i = 1; i < 5; i++)
{
    if (i == 3)
        break;

    Console.Write(i + " ");
}

Console.WriteLine();
```

## Array

```csharp
int[] arr = { 10, 20, 30, 40, 50 };

for (int i = 0; i < arr.Length; i++)
{
    Console.Write(arr[i] + " ");
}

Console.WriteLine();
```

## Foreach

```csharp
foreach (int value in arr)
{
    Console.Write(value + " ");
}

Console.WriteLine();
```

## 2D Array

```csharp
int[,] matrix =
{
    { 11, 12 },
    { 13, 14 },
    { 15, 16 }
};

int rows = matrix.GetLength(0);
int columns = matrix.GetLength(1);

for (int i = 0; i < rows; i++)
{
    for (int j = 0; j < columns; j++)
    {
        Console.Write(matrix[i, j] + " ");
    }

    Console.WriteLine();
}
```

## Exception Handling

```csharp
try
{
    Console.Write("Enter number 1: ");
    int num1 = Convert.ToInt32(Console.ReadLine());

    Console.Write("Enter number 2: ");
    int num2 = Convert.ToInt32(Console.ReadLine());

    Console.WriteLine(num1 / num2);
}
catch (DivideByZeroException)
{
    Console.WriteLine("Cannot divide by zero.");
}
catch (Exception e)
{
    Console.WriteLine(e.Message);
}
finally
{
    Console.WriteLine("Program completed.");
}
```

## Methods

### Calling Methods

```csharp
Fun("Abhi", 22);

int age = Convert.ToInt32(Console.ReadLine());

Console.WriteLine(Fun(age));
```

### Method Definitions

```csharp
static void Fun(string name, int age)
{
    Console.WriteLine("Name: " + name);
    Console.WriteLine("Age: " + age);
}

static int Fun(int age)
{
    return age + 10;
}
```
